cmake_minimum_required(VERSION 3.24)
project(ConnectClassroom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
option(CONNECT_LAB_SIMULATOR "Build generated-audio test executable instead of vendor runtime" OFF)
option(CONNECT_LAB_TEACHER_ANSWER "Use teacher implementation for the lesson task" OFF)
enable_testing()
add_executable(wav_tests tests/wav_tests.cpp)
target_include_directories(wav_tests PRIVATE src)
# Keep assert checks active in Release test builds.
if(MSVC)
  target_compile_options(wav_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(wav_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME core_wav COMMAND wav_tests)
add_executable(lesson_tests tests/lesson_tests.cpp)
if(CONNECT_LAB_TEACHER_ANSWER)
  target_include_directories(lesson_tests BEFORE PRIVATE answers)
endif()
target_include_directories(lesson_tests PRIVATE src)
add_test(NAME lesson_summary COMMAND lesson_tests)
add_library(ring_pulse STATIC src/RingPulse.cpp)
target_include_directories(ring_pulse PUBLIC src)
add_executable(ring_tests tests/ring_tests.cpp)
target_link_libraries(ring_tests PRIVATE ring_pulse)
add_test(NAME ring_pulse COMMAND ring_tests)
if(CONNECT_LAB_SIMULATOR)
  add_executable(connect_lab_simulated src/main.cpp tests/fake_sdk.cpp)
  set(lab_target connect_lab_simulated)
  target_compile_definitions(${lab_target} PRIVATE CONNECT_LAB_SIMULATED=1)
  if(WIN32)
    target_compile_definitions(${lab_target} PRIVATE VIAIM_CONNECT_SDK_BUILD=1)
  else()
    target_compile_definitions(${lab_target} PRIVATE __cdecl=)
  endif()
else()
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Vendor SDK requires Windows x64. For portable code checks use -DCONNECT_LAB_SIMULATOR=ON.")
  endif()
  add_executable(connect_lab src/main.cpp)
  set(lab_target connect_lab)
  set(CONNECT_VOICE_SDK_DIR "${CMAKE_BINARY_DIR}/voice-deps/sherpa_onnx"
    CACHE PATH "Sherpa ONNX package directory for optional voice snapshots")
  set(CONNECT_VOICE_MODEL_DIR
    "${CMAKE_BINARY_DIR}/voice-models/sherpa-onnx-sense-voice-zh-en-ja-ko-yue-int8-2024-07-17"
    CACHE PATH "Default local SenseVoice model directory")
  if(EXISTS "${CONNECT_VOICE_SDK_DIR}/include/sherpa-onnx/c-api/c-api.h"
     AND EXISTS "${CONNECT_VOICE_SDK_DIR}/lib/sherpa-onnx-c-api.lib"
     AND EXISTS "${CONNECT_VOICE_SDK_DIR}/lib/sherpa-onnx-c-api.dll"
     AND EXISTS "${CONNECT_VOICE_SDK_DIR}/lib/onnxruntime.dll")
    target_sources(connect_lab PRIVATE src/VoiceSnapshot.cpp)
    target_compile_definitions(connect_lab PRIVATE CONNECT_LAB_VOICE=1
      "CONNECT_VOICE_MODEL_DIR=\"${CONNECT_VOICE_MODEL_DIR}\"")
    target_include_directories(connect_lab PRIVATE "${CONNECT_VOICE_SDK_DIR}/include")
    target_link_libraries(connect_lab PRIVATE "${CONNECT_VOICE_SDK_DIR}/lib/sherpa-onnx-c-api.lib")
    foreach(file sherpa-onnx-c-api.dll onnxruntime.dll)
      add_custom_command(TARGET connect_lab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CONNECT_VOICE_SDK_DIR}/lib/${file}" "$<TARGET_FILE_DIR:connect_lab>")
    endforeach()
  endif()
  target_link_libraries(connect_lab PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/win64/viaim_connect_sdk.lib")
  foreach(file viaim_connect_sdk.dll opencv_world4120.dll viaim_connect_descriptions.json)
    add_custom_command(TARGET connect_lab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/vendor/win64/${file}" "$<TARGET_FILE_DIR:connect_lab>")
  endforeach()
endif()
if(WIN32)
  target_sources(${lab_target} PRIVATE src/VideoCapture.cpp src/RingMonitor.cpp src/RingOverlay.cpp)
  target_compile_definitions(${lab_target} PRIVATE CONNECT_LAB_VIDEO=1)
  target_link_libraries(${lab_target} PRIVATE ring_pulse winhttp windowsapp)
endif()
if(CONNECT_LAB_TEACHER_ANSWER)
  target_compile_definitions(${lab_target} PRIVATE CONNECT_LAB_TEACHER_ANSWER=1)
endif()
target_include_directories(${lab_target} PRIVATE vendor/include src)
